# 1. DATENBANK ERSTELLEN (CREATE)


CREATE DATABASE IF NOT EXISTS esports;
USE esports;

# Tabellen löschen (falls vorhanden)
DROP TABLE IF EXISTS matches;
DROP TABLE IF EXISTS tournaments;
DROP TABLE IF EXISTS players_teames;
DROP TABLE IF EXISTS team;
DROP TABLE IF EXISTS players;
DROP TABLE IF EXISTS games;

# Tabelle Games Erstellen
CREATE TABLE games (
    Game_id INT UNSIGNED AUTO_INCREMENT NOT NULL,
    G_Titel VARCHAR(30) NOT NULL,
    Genre VARCHAR(30) NOT NULL,
    Kurzbeschreibung TEXT(100) NULL DEFAULT 'Kein Beschreibung',
    PRIMARY KEY (Game_id)
);

# Tabelle Players Erstellen
CREATE TABLE players (
    Player_id INT UNSIGNED AUTO_INCREMENT NOT NULL,
    Name VARCHAR(30) NOT NULL,
    Vorname VARCHAR(30) NOT NULL,
    GebDatum DATE NOT NULL,
    Land VARCHAR(30) NOT NULL,
    Rolle VARCHAR(30) NOT NULL,
    Game_id INT UNSIGNED NOT NULL,
    PRIMARY KEY (Player_id),
    KEY index_players_Game_id (Game_id),
    CONSTRAINT fk_players_Games_id FOREIGN KEY (Game_id) REFERENCES games (Game_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);

# Tabelle Team Erstellen
CREATE TABLE team (
    Team_id INT UNSIGNED AUTO_INCREMENT NOT NULL,
    T_Name VARCHAR(30) NOT NULL,
    Game_id INT UNSIGNED NOT NULL,
    PRIMARY KEY (Team_id),
    KEY index_team_Game_id (Game_id),
    CONSTRAINT fk_team_Games_id FOREIGN KEY (Game_id) REFERENCES games (Game_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);

# Tabelle Players_Teames Erstellen
CREATE TABLE players_teames (
    Players_TeamesID INT UNSIGNED AUTO_INCREMENT NOT NULL,
    Player_id INT UNSIGNED NOT NULL,
    Team_id INT UNSIGNED NOT NULL,
    Game_id INT UNSIGNED NOT NULL,
    PRIMARY KEY (Players_TeamesID),
    CONSTRAINT fk_players_Teames_player_id FOREIGN KEY (Player_id) REFERENCES players (Player_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    KEY index_player_id (Player_id),
    CONSTRAINT fk_players_Team_id FOREIGN KEY (Team_id) REFERENCES team (Team_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    KEY index_team_id (Team_id),
    CONSTRAINT fk_players_teames_Games_id FOREIGN KEY (Game_id) REFERENCES games (Game_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    KEY index_players_teames_Game_id (Game_id)
);

# Tabelle Tournaments Erstellen
CREATE TABLE tournaments (
    Tournaments_id INT UNSIGNED AUTO_INCREMENT NOT NULL,
    Game_id INT UNSIGNED NOT NULL,
    T_name VARCHAR(30) NOT NULL,
    Datum DATE NOT NULL,
    Location VARCHAR(30) NOT NULL,
    prize_pool INT(10) NOT NULL DEFAULT 0,
    Winner_Team_id INT UNSIGNED NULL DEFAULT NULL,
    PRIMARY KEY (Tournaments_id),
    CONSTRAINT fk_tournaments_Game_id FOREIGN KEY (Game_id) REFERENCES games (Game_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_tournaments_winner FOREIGN KEY (Winner_Team_id) REFERENCES team (Team_id)
        ON UPDATE CASCADE ON DELETE SET NULL,
    KEY index_Game_id (Game_id),
    KEY index_winner_team (Winner_Team_id)
);

# Tabelle Matches Erstellen
CREATE TABLE matches (
    Match_id INT UNSIGNED AUTO_INCREMENT NOT NULL,
    Dauer_in_minuten INT UNSIGNED NOT NULL DEFAULT 0,
    Tournaments_id INT UNSIGNED NOT NULL,
    Score_Team1 INT UNSIGNED NOT NULL,
    Score_Team2 INT UNSIGNED NOT NULL,
    Team1_id INT UNSIGNED NOT NULL,
    Team2_id INT UNSIGNED NOT NULL,
    Winner_Team_id INT UNSIGNED NULL DEFAULT NULL,
    Game_id INT UNSIGNED NULL DEFAULT NULL,
    PRIMARY KEY (Match_id),
    CONSTRAINT fk_match_tournaments FOREIGN KEY (Tournaments_id) REFERENCES tournaments (Tournaments_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    KEY index_tournaments_id (Tournaments_id),
    CONSTRAINT fk_match_Team1_id FOREIGN KEY (Team1_id) REFERENCES team (Team_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    KEY index_team1_id (Team1_id),
    CONSTRAINT fk_match_Team2_id FOREIGN KEY (Team2_id) REFERENCES team (Team_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    KEY index_Team2_id (Team2_id),
    CONSTRAINT fk_match_winner FOREIGN KEY (Winner_Team_id) REFERENCES team (Team_id)
        ON UPDATE CASCADE ON DELETE SET NULL,
    KEY index_winner_team (Winner_Team_id),
    CONSTRAINT fk_matches_Game_id FOREIGN KEY (Game_id) REFERENCES games (Game_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
);



# 2. TRIGGER ERSTELLEN


DROP TRIGGER IF EXISTS check_team_size;

DELIMITER //

CREATE TRIGGER check_team_size
BEFORE INSERT ON players_teames
FOR EACH ROW
BEGIN
    DECLARE player_count INT;

    SELECT COUNT(*) INTO player_count
    FROM players_teames
    WHERE Team_id = NEW.Team_id AND Game_id = NEW.Game_id;

    IF player_count >= 5 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Ein Team darf maximal 5 Spieler haben';
    END IF;
END;
//

DELIMITER ;



# 3. VIEW ERSTELLEN


CREATE OR REPLACE VIEW view_team_size AS
SELECT
    t.T_Name AS Team,
    g.G_Titel AS Spiel,
    COUNT(pt.Player_id) AS Spieleranzahl
FROM players_teames pt
JOIN team t ON pt.Team_id = t.Team_id
JOIN games g ON pt.Game_id = g.Game_id
GROUP BY pt.Team_id, pt.Game_id;

# Spieler 'Justin Shanks' komplett aus players_teames entfernen
DELETE pt
FROM players_teames pt
JOIN players p ON pt.Player_id = p.Player_id
WHERE p.Name = 'Justin' AND p.Vorname = 'Shanks' AND p.Game_id = 3;

# 8. DATEN ABFRAGEN (SELECT)


# 1. Alle Spieler aus League of Legends sortiert nach Vorname
SELECT Name, Vorname, Rolle, Land
FROM players
WHERE Game_id = 1
ORDER BY Vorname;

# 2. Alle Turniere mit Prize Pool >= 1.000.000 $
SELECT Game_id, T_name, Location, prize_pool
FROM tournaments
WHERE prize_pool >= 1000000
ORDER BY prize_pool DESC;

# 3. Spieler mit Team – nur Valorant (1:n Beziehung)
SELECT p.Name, p.Vorname, t.T_Name AS Team, g.G_Titel AS Spiel
FROM players p
JOIN players_teames pt ON p.Player_id = pt.Player_id
JOIN team t ON pt.Team_id = t.Team_id
JOIN games g ON pt.Game_id = g.Game_id
WHERE g.G_Titel = 'Valorant'
ORDER BY t.T_Name, p.Vorname;

# 4. Alle Spieler mit Team und Spiel (n:m Beziehung)
SELECT p.Name, p.Vorname, t.T_Name AS Team, g.G_Titel AS Spiel, p.Rolle
FROM players p
JOIN players_teames pt ON p.Player_id = pt.Player_id
JOIN team t ON pt.Team_id = t.Team_id
JOIN games g ON pt.Game_id = g.Game_id
ORDER BY g.G_Titel, t.T_Name, p.Vorname;
